# Хранилище: список ссылок, которые прошли проверку и опубликованы

def clean_queue():
    """Оставляет в очереди только последние MAX_QUEUE_SIZE ссылок.
       Удаляет самые старые, если очередь переполнена."""
    global queue
    if len(queue) > MAX_QUEUE_SIZE:
        # Удаляем первые (самые старые) элементы
        removed = queue[:-MAX_QUEUE_SIZE]
        queue = queue[-MAX_QUEUE_SIZE:]
        
        # Очищаем pending_links для пользователей, чьи ссылки были удалены
        removed_links = [item['link'] for item in removed]
        for user_id in list(pending_links.keys()):
            pending_links[user_id] = [link for link in pending_links[user_id] if link not in removed_links]
            if not pending_links[user_id]:
                del pending_links[user_id]
        
        print(f"🧹 Очередь очищена. Удалено {len(removed)} старых ссылок. Теперь в очереди: {len(queue)}")

def extract_wall_id(text):
    """Извлекает из текста ID поста вида wall-123_456"""
    match = re.search(r'wall(-?\d+)_(\d+)', text)
    return match.group(0) if match else None

def check_post_exists(wall_id):
    """Проверяет, существует ли пост с таким wall_id"""
    parts = wall_id.replace('wall', '').split('_')
    owner_id = int(parts[0])
    post_id = int(parts[1])
    
    try:
        vk.wall.getById(posts=f"{owner_id}_{post_id}")
        return True
    except Exception as e:
        error_msg = str(e)
        if 'access denied' in error_msg.lower():
            return False
        elif 'not found' in error_msg.lower():
            return False
        else:
            return False

def check_like(user_id, wall_id):
    """Проверяет, поставил ли пользователь лайк на указанный пост."""
    
    parts = wall_id.replace('wall', '').split('_')
    owner_id = int(parts[0])
    item_id = int(parts[1])
    
    try:
        response = vk.likes.isLiked(
            user_id=user_id,
            type='post',
            owner_id=owner_id,
            item_id=item_id
        )
        if isinstance(response, dict):
            return response.get('liked', 0) == 1
        else:
            return response == 1
    except Exception as e:
        print(f"Ошибка проверки лайка: {e}")
        return False

def check_previous_likes(user_id):
    """Проверяет, поставил ли пользователь лайки на последние 10 ссылок в очереди."""
    
    links_to_check = [item['link'] for item in queue[-10:]]
    
    if not links_to_check:
        return True, []
    
    missing = []
    for link in links_to_check:
        if not check_like(user_id, link):
            missing.append(link)
    
    return len(missing) == 0, missing

def can_user_post(user_id):
    """Проверяет, можно ли пользователю отправить ссылку.
       Условие: после его предыдущей ссылки прошло 5 новых чужих ссылок."""
    
    last_index = -1
    for i, item in enumerate(queue):
        if item['user_id'] == user_id:
            last_index = i
    
    if last_index == -1:
        return True
    
    if last_index >= len(queue) - 1:
        return False
posts_after = len(queue) - last_index - 1
    return posts_after >= 5

def send_message(peer_id, text, user_id=None):
    """Отправляет сообщение в чат и сохраняет его ID для возможного удаления"""
    result = vk.messages.send(
        peer_id=peer_id,
        message=text,
        random_id=int(time.time() * 1000)
    )
    
    # Если сообщение отправлено пользователю (не боту) и это служебное сообщение
    if user_id and user_id > 0:
        if user_id not in bot_messages:
            bot_messages[user_id] = []
        bot_messages[user_id].append(result)
    
    return result

def delete_message(peer_id, message_id):
    """Удаляет сообщение"""
    try:
        vk.messages.delete(
            peer_id=peer_id,
            message_ids=[message_id],
            delete_for_all=True
        )
        return True
    except Exception as e:
        print(f"Ошибка удаления сообщения: {e}")
        return False

def delete_bot_messages(user_id, peer_id):
    """Удаляет все служебные сообщения бота для конкретного пользователя"""
    if user_id in bot_messages:
        for msg_id in bot_messages[user_id]:
            delete_message(peer_id, msg_id)
        del bot_messages[user_id]
        print(f"🗑️ Удалены служебные сообщения для пользователя {user_id}")

def process_message(event):
    """Обрабатывает новое сообщение"""
    peer_id = event.object.message['peer_id']
    user_id = event.object.message['from_id']
    text = event.object.message.get('text', '').strip()
    message_id = event.object.message['id']
    
    # Игнорируем сообщения от самого бота
    if user_id < 0:
        return
    
    # Проверяем, есть ли в сообщении ссылка на пост
    wall_id = extract_wall_id(text)
    
    # Если ссылки нет — удаляем сообщение и пишем в чат
    if not wall_id:
        delete_message(peer_id, message_id)
        send_message(
            peer_id,
            f"❌ @id{user_id}, для публикации нужна ссылка на открытый пост ВК.\n"
            f"Формат: wall-123_456",
            user_id=user_id
        )
        return
    
    # 0. Проверяем, существует ли пост и открыт ли он
    if not check_post_exists(wall_id):
        delete_message(peer_id, message_id)
        send_message(
            peer_id,
            f"❌ @id{user_id}, пост {wall_id} не найден или закрыт.\n"
            f"Нужна ссылка на открытый пост ВК.",
            user_id=user_id
        )
        return
    
    # 1. Проверяем условие "своя ссылка раз в 5 чужих"
    if not can_user_post(user_id):
        last_index = -1
        for i, item in enumerate(queue):
            if item['user_id'] == user_id:
                last_index = i
        
        posts_after = len(queue) - last_index - 1
        need_to_wait = 5 - posts_after
        
        delete_message(peer_id, message_id)
        
        send_message(
            peer_id,
            f"❌ @id{user_id}, ты можешь отправить новую ссылку только после {need_to_wait} чужих постов.\n"
            f"Сейчас после твоей последней ссылки прошло {posts_after} постов.",
            user_id=user_id
        )
        return
    
    # 2. Проверяем лайки на предыдущие 10 ссылок
    all_liked, missing_links = check_previous_likes(user_id)
    
    if not all_liked:
        pending_links[user_id] = missing_links
        
        delete_message(peer_id, message_id)
        
        missing_text = "\n".join([f"• {link}" for link in missing_links])
        send_message(
            peer_id,
            f"❌ @id{user_id}, ты пропустил лайки на эти посты:\n{missing_text}\n\n"
            f"📌 Поставь лайки на все пропущенные посты и отправь ссылку заново.\n"
            f"Твоё сообщение со ссылкой удалено до выполнения условий.",
            user_id=user_id
        )
        return
    
    # 3. ВСЕ УСЛОВИЯ ВЫПОЛНЕНЫ — публикуем ссылку
    queue.append({
        'link': wall_id,
        'user_id': user_id,
        'timestamp': datetime.now()
    })
    
    # Очищаем очередь, оставляя только последние MAX_QUEUE_SIZE ссылок
clean_queue()
    
    # УДАЛЯЕМ ВСЕ СЛУЖЕБНЫЕ СООБЩЕНИЯ БОТА ДЛЯ ЭТОГО ПОЛЬЗОВАТЕЛЯ
    delete_bot_messages(user_id, peer_id)
    
    # Если у пользователя были пропущенные лайки — очищаем
    if user_id in pending_links:
        del pending_links[user_id]
    
    # Рассчитываем, через сколько ссылок пользователь сможет отправить новую
    last_index = -1
    for i, item in enumerate(queue):
        if item['user_id'] == user_id:
            last_index = i
    
    if last_index < len(queue) - 1:
        posts_after = len(queue) - last_index - 1
        wait_for = max(0, 5 - posts_after)
        if wait_for == 0:
            extra_msg = "Ты уже можешь отправить следующую ссылку!"
        else:
            extra_msg = f"Осталось ждать {wait_for} чужих постов."
    else:
        extra_msg = "Твоя ссылка последняя в очереди. Жди чужих постов."
    
    # Отправляем финальное подтверждение (оно НЕ удаляется)
    send_message(
        peer_id,
        f"✅ @id{user_id}, всё выполнено! Твоя ссылка принята.\n"
        f"📊 В очереди: {len(queue)} постов (хранятся последние {MAX_QUEUE_SIZE})\n"
        f"⏳ {extra_msg}"
        # НЕ передаём user_id, чтобы это сообщение не попало в список на удаление
    )
